def exe():
    board = pcbnew.GetBoard()
    print("Targeting active PCB File for 24.50mm x 20.00mm Placement in KiCad 10...")

    OFFSET_X = 100.0
    OFFSET_Y = 100.0
    BOARD_W = 24.50
    BOARD_H = 20.00

    # 1. Clear Edge.Cuts drawings safely
    try:
        for d in list(board.GetDrawings()):
            if d.GetLayerName() == "Edge.Cuts":
                board.Delete(d)
    except Exception as e:
        print(f"Cleanup note: {e}")

    try:
        edge_layer = board.GetLayerID("Edge.Cuts")
    except Exception:
        edge_layer = pcbnew.Edge_Cuts

    # 2. Draw centered Edge.Cuts boundary rectangle
    rect_pts = [
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y + BOARD_H)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y + BOARD_H)),
    ]

    for i in range(4):
        seg = pcbnew.PCB_SHAPE(board)
        seg.SetShape(pcbnew.SHAPE_T_SEGMENT if hasattr(pcbnew, 'SHAPE_T_SEGMENT') else pcbnew.S_SEGMENT)
        seg.SetStart(rect_pts[i])
        seg.SetEnd(rect_pts[(i + 1) % 4])
        seg.SetLayer(edge_layer)
        seg.SetWidth(pcbnew.FromMM(0.15))
        board.Add(seg)

    def find_fp(ref):
        return board.FindFootprintByReference(ref)

    def place_footprint(ref, x_mm, y_mm, angle_deg, flip_bottom=False):
        fp = find_fp(ref)
        if not fp:
            print(f"Warning: Component '{ref}' NOT FOUND.")
            return

        px = OFFSET_X + x_mm
        py = OFFSET_Y + y_mm
        pos_vec = pcbnew.VECTOR2I(pcbnew.FromMM(px), pcbnew.FromMM(py))

        is_on_bottom = (fp.GetLayer() == pcbnew.B_Cu)
        if flip_bottom and not is_on_bottom:
            fp.Flip(pos_vec, False)
        elif not flip_bottom and is_on_bottom:
            fp.Flip(pos_vec, False)

        fp.SetPosition(pos_vec)
        fp.SetOrientationDegrees(angle_deg)

    # Footprint Placement Coordinates
    place_footprint("U1", 9.50, 10.00, 0)                    # ESP32-C6 MINI-1 (Top)
    place_footprint("J1", 20.80, 13.50, 90)                  # USB-C (Right Edge)
    place_footprint("J2", 17.00, 4.50, 0, flip_bottom=True)  # SMT JST-PH Connector (B.Cu)
    place_footprint("R3", 20.50, 6.50, 0)                    # CC1 Pull-down
    place_footprint("R4", 20.50, 8.00, 0)                    # CC2 Pull-down
    place_footprint("U3", 9.00, 18.20, 0)                    # MCP73831 Charger
    place_footprint("R5", 13.50, 18.20, 0)                   # PROG Resistor
    place_footprint("U4", 3.20, 2.80, 0)                     # AP2112K LDO
    place_footprint("C1", 6.20, 2.80, 90)                    # Decoupling Cap
    place_footprint("J3", 1.20, 10.50, 0)                    # 4-Pin Header
    place_footprint("R1", 1.20, 5.00, 90)                    # I2C SDA Pull-up
    place_footprint("R2", 1.20, 7.00, 90)                    # I2C SCL Pull-up
    place_footprint("D1", 4.00, 18.20, 180)                  # BAT54C Diode

import pcbnew

def exe():
    board = pcbnew.GetBoard()
    OFFSET_X, OFFSET_Y = 100.0, 100.0
    BOARD_W, BOARD_H = 24.50, 20.00

    def get_net(net_name):
        net = board.FindNet(net_name)
        if not net or net.GetNetname() == "":
            net = board.FindNet("/" + net_name)
        return net

    gnd_net = get_net("GND")
    zone = pcbnew.ZONE(board)
    zone.SetLayer(pcbnew.B_Cu)
    if gnd_net:
        zone.SetNet(gnd_net)

    outline = pcbnew.SHAPE_LINE_CHAIN()
    outline.Append(pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y)))
    outline.Append(pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y)))
    outline.Append(pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y + BOARD_H)))
    outline.Append(pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y + BOARD_H)))
    outline.SetClosed(True)

    try:
        zone.AddPolygon(outline)
    except Exception:
        zone.Outline().AddOutline(outline)

    board.Add(zone)

    try:
        filler = pcbnew.ZONE_FILLER(board)
        filler.Fill(board.Zones())
    except Exception:
        pass

    pcbnew.Refresh()
    print("Ground zone established on B.Cu.")

exe()

## 5. HARDWARE FMEA, RISK MITIGATION & REGULATORY PATHWAY

### Failure Mode & Effects Analysis (FMEA)

1. **Reverse Battery Polarity Protection**: P-FET (e.g. DMG2305CX) or Schottky diode placed on J2 battery line to prevent immediate destruction of U3/U4 if battery lead polarity is reversed.
2. **Antenna Keepout Zone**: Zero copper/traces/vias on any layer within the 3.0 mm clearance zone surrounding the ESP32-C6 PCB trace antenna to maintain 50-ohm impedance.
3. **Sensor Thermal Isolation**: Milled PCB slots / cutouts on the satellite daughterboard around the BME688 to prevent heat conduction from MCU and LDO skewing temperature and relative humidity readings.
4. **I2C Bus Floating / Address Lockup**: Hardwire BME688 `CSB` pin to 3.3V and `SDO` pin to GND on daughterboard layout. Keep 4.7 kOhm pull-up resistors solely on the main carrier board.

### Regulatory Certification Strategy (FCC / CE)

* **RF Module Modular Grant**: The ESP32-C6-MINI-1 holds full FCC Modular Approval (`FCC ID: 2AC7Z-ESPC6MINI1`). Bypasses expensive intentional radiator RF testing ($10,000 - $25,000).
* **FCC Part 15 Subpart B**: Required for commercial launch. Tests unintentional radiated emissions from digital traces, USB lines, and voltage regulators (~$1,500 - $2,500).
* **CE Marking (EU)**: EMC Directive 2014/30/EU and RoHS compliance required. Managed via self-declarationbacked by lab EMC reports (~$2,000 - $3,500).

* **Idle Mode**: Boots into `MODE_IDLE` using `PROFILE_OFF` / `PROFILE_ULP_300S`[cite: 5].
* **Light Sleep**: `MODE_IDLE` invokes `enterLightSleep(100)` (100 ms timer ticks) to keep the CPU dormant while maintaining SRAM state and BLE radio responsiveness[cite: 5].
* **Warmup Routine**: Transitioning out of `MODE_IDLE` executes `performWarmup()` (`PROFILE_LP_3S` for 4 seconds) to re-stabilize sensor heater registers before active sampling[cite: 5].

### BLE Command Protocol
* `"1"`, `"r"`, `"run"` -> Sets `PROFILE_LP_3S` (3-second LP mode), enters `MODE_DRY_AIR_DETECTION`[cite: 5].
* `"set_active_sampling_mode"`, `"set_lp"` -> Switches to `PROFILE_LP_3S`[cite: 5].
* `"set_ultra_low_sampling_mode"`, `"set_ulp"` -> Switches to `PROFILE_ULP_300S` (5-minute ULP mode)[cite: 5].
* `"2"`, `"s"`, `"stop"` -> Sets `PROFILE_OFF`, halts dry air mode, returns CPU to `MODE_IDLE`[cite: 5].
* `"3"`, `"b"`, `"breath"` -> Triggers `runBreathSequence()`: performs thermal warmup, prompts `READY_PLEASE_BLOW`, captures VOC nadir/RH spike over an active sensing window, evaluates clinical score (`FRESH`, `MILD`, `SIGNIFICANT`, `SEVERE`), and dispatches final result[cite: 5].
* `"10"`, `"status"` -> Returns current operational JSON state[cite: 5].

### Offline Data Storage & Reconnection Synchronization
* **Live Connection**: Dispatches formatted JSON payloads over NimBLE notifications[cite: 5].
* **Offline Mode**: When disconnected, sensor readings accumulate in SRAM and write 5-minute averaged entries (300,000 ms) to LittleFS (`/offline_data.json`) to prevent flash wear[cite: 5].
* **Asynchronous Reconnect Flush**: `ServerCallbacks::onConnect()` sets `pendingFlush = true`[cite: 5]. Main `loop()` detects this flag, streams cached JSON lines over BLE via `flushOfflineData()`, and erases the file[cite: 5].

### Resolved Bug Log (Breath Test Stalling Issue - Fixed in Last 24 Hours)
* **Symptom**: Triggering `breath` sent `READY_PLEASE_BLOW` but immediately dropped back to `PROFILE_LP` and printed `TIMEOUT` without streaming real-time `{"dH":...}` telemetry packets.
* **Root Causes**:
  1. `bsecReady` evaluated to `false` or BSEC failed during profile switch to `PROFILE_CONT_1S`.
  2. `newGasDataAvailable` never flipped `true` in time when switching to continuous 1s mode, causing `waitAndCaptureBreath()` to stall, exit early, and trigger `TIMEOUT` inside `runBreathSequence()`.
* **Verified Resolution (`waitAndCaptureBreath()`)**:

```cpp
bool waitAndCaptureBreath(float &outBaselineRes, float &outMinRes) {
  if (!bsecReady) {
    logMessage("[ERROR]: waitAndCaptureBreath aborted because bsecReady is false!");
    return false;
  }

  // 1. Switch sensor to 1-second continuous sampling mode
  setBsecProfile(PROFILE_CONT_1S);
  
  // Give BSEC time to initialize continuous mode
  vTaskDelay(pdMS_TO_TICKS(100));

  // 2. Wait for BSEC to produce at least 1 valid continuous frame
  newGasDataAvailable = false;
  uint32_t stabStart = millis();
  
  while (!newGasDataAvailable) {
    bsec.run();
    vTaskDelay(pdMS_TO_TICKS(POLL_TICK_DELAY_MS));
    
    // Fallback: If BSEC stalls on profile switch, force proceed after 2.5s using current cached values
    if (millis() - stabStart > 2500UL) {
      logMessage("[WARNING]: BSEC gas frame delay during CONT switch. Proceeding with cached baseline.");
      break; 
    }
  }

  // Lock in baseline values
  newGasDataAvailable = false;
  float baseHumidity = sensorData.currentHumidity;
  float baseGasRes   = sensorData.currentGasRes;
  outBaselineRes     = baseGasRes;
  outMinRes          = baseGasRes;

  dispatchData("{\"state\":\"READY_PLEASE_BLOW\"}");
  vTaskDelay(pdMS_TO_TICKS(50)); 

  // 3. Start the 10-second breath wait window
  uint32_t startMs = millis();
  bool detected = false;
  newGasDataAvailable = false;

  while ((millis() - startMs) < BREATH_WAIT_TIMEOUT_MS) {
    bsec.run();

    if (newGasDataAvailable) {
      newGasDataAvailable = false;

      float deltaHumidity = sensorData.currentHumidity - baseHumidity;
      float gasDropPct = (baseGasRes > 0.0f && sensorData.currentGasRes > 0.0f) ? 
                         ((baseGasRes - sensorData.currentGasRes) / baseGasRes) * 100.0f : 0.0f;

      if (sensorData.currentGasRes > 0.0f && sensorData.currentGasRes < outMinRes) {
        outMinRes = sensorData.currentGasRes;
      }

      dispatchData("{\"dH\":" + String(deltaHumidity, 1) + ",\"gDrop\":" + String(gasDropPct, 1) + "}");

      // Trigger if user exhales (RH jump >= 0.8% OR VOC drop >= 1.5%)
      if (deltaHumidity >= 0.8f || gasDropPct >= 1.5f) {
        detected = true;
        break;
      }
    }
    vTaskDelay(pdMS_TO_TICKS(POLL_TICK_DELAY_MS));
  }

  if (!detected) {
    setBsecProfile(PROFILE_LP_3S);
    return false;
  }

  // 4. Hold analyzing state for 5 full seconds while capturing nadir VOC resistance
  dispatchData("{\"state\":\"TESTING_SENSING_BREATH\"}");
  vTaskDelay(pdMS_TO_TICKS(50));

  uint32_t blowWindowStart = millis();
  newGasDataAvailable = false;

  while (millis() - blowWindowStart < BREATH_SENSING_WINDOW_MS) {
    bsec.run();
    if (newGasDataAvailable) {
      newGasDataAvailable = false;
      if (sensorData.currentGasRes > 0.0f && sensorData.currentGasRes < outMinRes) {
        outMinRes = sensorData.currentGasRes; 
      }
    }
    vTaskDelay(pdMS_TO_TICKS(POLL_TICK_DELAY_MS));
  }

  setBsecProfile(PROFILE_LP_3S);
  return true;
}

import pcbnew

def exe():
    board = pcbnew.GetBoard()
    print("Targeting active PCB File for 24.50mm x 20.00mm Placement in KiCad 10...")

    OFFSET_X = 100.0
    OFFSET_Y = 100.0
    BOARD_W = 24.50
    BOARD_H = 20.00

    # 1. Clear Edge.Cuts drawings safely
    try:
        for d in list(board.GetDrawings()):
            if d.GetLayerName() == "Edge.Cuts":
                board.Delete(d)
    except Exception as e:
        print(f"Cleanup note: {e}")

    try:
        edge_layer = board.GetLayerID("Edge.Cuts")
    except Exception:
        edge_layer = pcbnew.Edge_Cuts

    # 2. Draw centered Edge.Cuts boundary rectangle
    rect_pts = [
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X + BOARD_W), pcbnew.FromMM(OFFSET_Y + BOARD_H)),
        pcbnew.VECTOR2I(pcbnew.FromMM(OFFSET_X), pcbnew.FromMM(OFFSET_Y + BOARD_H)),
    ]

    for i in range(4):
        seg = pcbnew.PCB_SHAPE(board)
        seg.SetShape(pcbnew.SHAPE_T_SEGMENT if hasattr(pcbnew, 'SHAPE_T_SEGMENT') else pcbnew.S_SEGMENT)
        seg.SetStart(rect_pts[i])
        seg.SetEnd(rect_pts[(i + 1) % 4])
        seg.SetLayer(edge_layer)
        seg.SetWidth(pcbnew.FromMM(0.15))
        board.Add(seg)

    def find_fp(ref):
        return board.FindFootprintByReference(ref)

    def place_footprint(ref, x_mm, y_mm, angle_deg, flip_bottom=False):
        fp = find_fp(ref)
        if not fp:
            print(f"Warning: Component '{ref}' NOT FOUND.")
            return

        px = OFFSET_X + x_mm
        py = OFFSET_Y + y_mm
        pos_vec = pcbnew.VECTOR2I(pcbnew.FromMM(px), pcbnew.FromMM(py))

        is_on_bottom = (fp.GetLayer() == pcbnew.B_Cu)
        if flip_bottom and not is_on_bottom:
            fp.Flip(pos_vec, False)
        elif not flip_bottom and is_on_bottom:
            fp.Flip(pos_vec, False)

        fp.SetPosition(pos_vec)
        fp.SetOrientationDegrees(angle_deg)

    # Footprint Placement Coordinates
    place_footprint("U1", 9.50, 10.00, 0)                    # ESP32-C6 MINI-1 (Top)
    place_footprint("J1", 20.80, 13.50, 90)                  # USB-C (Right Edge)
    place_footprint("J2", 17.00, 4.50, 0, flip_bottom=True)  # SMT JST-PH Connector (B.Cu)
    place_footprint("R3", 20.50, 6.50, 0)                    # CC1 Pull-down
    place_footprint("R4", 20.50, 8.00, 0)                    # CC2 Pull-down
    place_footprint("U3", 9.00, 18.20, 0)                    # MCP73831 Charger
    place_footprint("R5", 13.50, 18.20, 0)                   # PROG Resistor
    place_footprint("U4", 3.20, 2.80, 0)                     # AP2112K LDO
    place_footprint("C1", 6.20, 2.80, 90)                    # Decoupling Cap
    place_footprint("J3", 1.20, 10.50, 0)                    # 4-Pin Header
    place_footprint("R1", 1.20, 5.00, 90)                    # I2C SDA Pull-up
    place_footprint("R2", 1.20, 7.00, 90)                    # I2C SCL Pull-up
    place_footprint("D1", 4.00, 18.20, 180)                  # BAT54C Diode

    pcbnew.Refresh()
    print("Placement script complete.")

def exe():
    board = pcbnew.GetBoard()
    OFFSET_X, OFFSET_Y = 100.0, 100.0
    BOARD_W, BOARD_H = 24.50, 20.00

    def get_net(net_name):

    gnd_net = get_net("GND")
    zone = pcbnew.ZONE(board)
    zone.SetLayer(pcbnew.B_Cu)

    outline.SetClosed(True)

    try:
        zone.AddPolygon(outline)
    except Exception:
        zone.Outline().AddOutline(outline)

    try:
        filler = pcbnew.ZONE_FILLER(board)

## 5. HARDWARE FMEA, RISK MITIGATION & REGULATORY PATHWAY

### Failure Mode & Effects Analysis (FMEA)

1. **Reverse Battery Polarity Protection**: P-FET (e.g. DMG2305CX) or Schottky diode placed on J2 battery line to prevent immediate destruction of U3/U4 if battery lead polarity is reversed.
2. **Antenna Keepout Zone**: Zero copper/traces/vias on any layer within the 3.0 mm clearance zone surrounding the ESP32-C6 PCB trace antenna to maintain 50-ohm impedance.
3. **Sensor Thermal Isolation**: Milled PCB slots / cutouts on the satellite daughterboard around the BME688 to prevent heat conduction from MCU and LDO skewing temperature and relative humidity readings.
4. **I2C Bus Floating / Address Lockup**: Hardwire BME688 `CSB` pin to 3.3V and `SDO` pin to GND on daughterboard layout. Keep 4.7 kOhm pull-up resistors solely on the main carrier board.

### Regulatory Certification Strategy (FCC / CE)

* **RF Module Modular Grant**: The ESP32-C6-MINI-1 holds full FCC Modular Approval (`FCC ID: 2AC7Z-ESPC6MINI1`). Bypasses expensive intentional radiator RF testing ($10,000 - $25,000).


* **FCC Part 15 Subpart B**: Required for commercial launch. Tests unintentional radiated emissions from digital traces, USB lines, and voltage regulators (~$1,500 - $2,500).
* **CE Marking (EU)**: EMC Directive 2014/30/EU and RoHS compliance required. Managed via self-declaration backed by lab EMC reports (~$2,000 - $3,500).
